const [cart, setCart] = useState([]);

  useEffect(() => {
    axios.get("/products.json").then((data) => {
      const products = data.data;
      let items = JSON.parse(localStorage.getItem("cart")) || [];
      const result = products.filter((product) => items.includes(product.id));
      setCart(result);
    });
  }, []);

  const handleAddCart = (id, product) => {
    const existProduct = cart.find((c) => c.id === id);
    if (existProduct) {
      return alert("You have already Add This Product");
    }
    const updateCart = [...cart, product];
    setCart(updateCart);
    // LocalStorage Cart

    let items = JSON.parse(localStorage.getItem("cart")) || [];

    // const existItem = items.find((itemId) => itemId === id);
    const updateItems = [...items, id];
    localStorage.setItem("cart", JSON.stringify(updateItems));
  };

  const [cart, setCart] = useState([]);
  const [products, setProducts] = useState([]);

  // Products load
  useEffect(() => {
    axios.get("/products.json").then((res) => {
      setProducts(res.data);
    });
  }, []);

  // LocalStorage থেকে cart IDs নিয়ে product বের করা
  useEffect(() => {
    const cartIds = JSON.parse(localStorage.getItem("cart")) || [];

    const result = products.filter((product) =>
      cartIds.includes(product.id)
    );

    setCart(result);
  }, [products]);

  // Add to cart
  const handleAddCart = (id) => {
    const cartIds = JSON.parse(localStorage.getItem("cart")) || [];

    // Already added?
    if (cartIds.includes(id)) {
      return Swal.fire("You have already added this Product");
    }

    const updateCartIds = [...cartIds, id];

    // LocalStorage update
    localStorage.setItem("cart", JSON.stringify(updateCartIds));

    // সাথে সাথে React state update
    const addedProduct = products.find((product) => product.id === id);

    setCart((prevCart) => [...prevCart, addedProduct]);
  };
